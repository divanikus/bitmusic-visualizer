# Standalone replacement-library build shipped with the portable package.
cmake_minimum_required(VERSION 3.24)
project(BitMusicGme LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
include(FetchContent)
set(GME_BUILD_SHARED ON CACHE BOOL "" FORCE)
set(GME_BUILD_STATIC OFF CACHE BOOL "" FORCE)
set(GME_BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(GME_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GME_ZLIB OFF CACHE BOOL "" FORCE)
set(GME_YM2612_EMU Nuked CACHE STRING "" FORCE)
FetchContent_Declare(gme URL "${CMAKE_CURRENT_SOURCE_DIR}/../libgme-0.6.5.tar.gz"
  URL_HASH SHA256=8531678502451c2cf04248cda45c8b4645e19fcfb63e6a7ec2549641c47bb392 DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(gme)
include(cmake/PatchGme.cmake)
